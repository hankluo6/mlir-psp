cmake_minimum_required(VERSION 3.20)
project(CoffAddr32NB LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(LLVM REQUIRED CONFIG)
add_executable(coff-addr32nb-harness harness.cpp)
target_include_directories(coff-addr32nb-harness PRIVATE ${LLVM_INCLUDE_DIRS})
target_compile_definitions(coff-addr32nb-harness PRIVATE ${LLVM_DEFINITIONS})
llvm_map_components_to_libnames(LLVM_LIBS ExecutionEngine RuntimeDyld Object Support)
target_link_libraries(coff-addr32nb-harness PRIVATE ${LLVM_LIBS})
